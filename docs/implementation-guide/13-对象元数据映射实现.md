# 13 — 对象元数据映射实现（伪代码→可编译 C）

> **这是把文档 05 的伪代码翻译成真实 C 代码的关键文档。** 文档 05 中出现的 `rc_node(obj)`、`find_node(state, obj)`、`forward[obj]`、`forwarded(obj)` 等"魔法函数"，在这里给出**完整可编译实现**。低能力 AI 必须用这里的代码，不要自己发明。

---

## 1. 核心问题：从 gc_header* 到算法私有节点

文档 05 的伪代码大量出现 `rc_node(obj)->rc++`、`find_node(state, obj)->marked = 1`。问题是：

- `gc_header` 只有 4 个字段（desc/size/flags/kind），**没有 rc / marked / color 字段**
- 算法的私有状态（rc、marked、color、forwarding）必须存在**别处**

有三种映射方案，按场景选用。

---

## 2. 方案 A：内联头法（marksweep / rc-cycle 用）

**思想**：分配对象时，在 `gc_header` **前面**多分配一段算法私有头。对象指针和私有头通过固定偏移互转。

### 2.1 内存布局

```
低地址                                          高地址
┌────────────────┬──────────────────────────────────┐
│ 算法私有头      │ gc_header + payload                │
│ (GcMsMeta)     │ (用户可见的对象)                    │
└────────────────┴──────────────────────────────────┘
        ▲                  ▲
        │                  │
   meta 指针          obj 指针（返回给用户）
```

### 2.2 marksweep 的完整实现

```c
/* src/algo/marksweep/ms.c */

/* 算法私有头 — 放在对象内存的前面 */
typedef struct GcMsMeta {
    uint8_t           marked;
    struct GcMsMeta*  next;       /* 全对象链表 */
    /* 注意：对齐到 16 字节，保证后面的 gc_header 对齐 */
    uint8_t           _pad[16 - sizeof(uint8_t) - sizeof(void*)];
} GcMsMeta;

/* obj → meta：向前偏移 */
static inline GcMsMeta* ms_meta_of(gc_header* obj) {
    return (GcMsMeta*)((uint8_t*)obj - sizeof(GcMsMeta));
}

/* meta → obj：向后偏移 */
static inline gc_header* ms_obj_of(GcMsMeta* meta) {
    return (gc_header*)((uint8_t*)meta + sizeof(GcMsMeta));
}

/* 分配：一次性分配 meta + 对象 */
static void* gc_marksweep_alloc(gc_runtime* rt, gc_thread_context* thread,
                                const gc_descriptor* desc, size_t size, uint32_t flags) {
    (void)thread; (void)flags;
    GcMarksweepState* state = (GcMarksweepState*)rt->algo_state;

    /* 一块连续内存：[GcMsMeta][gc_header + payload] */
    void* block = calloc(1, sizeof(GcMsMeta) + size);
    if (!block) return NULL;

    GcMsMeta* meta = (GcMsMeta*)block;
    gc_header* obj = ms_obj_of(meta);

    obj->desc = desc;
    obj->size = (uint32_t)size;
    obj->kind = desc ? desc->kind : 0;

    /* 挂到全对象链表 */
    meta->marked = 0;
    meta->next   = state->objects;
    state->objects = meta;

    rt->stats.total_allocated += size;
    if (rt->stats.total_allocated > rt->stats.peak_allocated)
        rt->stats.peak_allocated = rt->stats.total_allocated;

    return obj;
}

/* find_node 不再需要 — 直接 ms_meta_of(obj) 即可 O(1) 反查！ */
/* 文档 05 中的 find_node(state, obj) 全部替换为 ms_meta_of(obj) */

/* mark 回调 */
static void ms_mark_callback(gc_header** slot, void* ctx) {
    GcMarksweepMarkCtx* c = (GcMarksweepMarkCtx*)ctx;
    gc_header* child = *slot;
    if (!child) return;
    GcMsMeta* meta = ms_meta_of(child);
    if (!meta->marked) {
        meta->marked = 1;
        gc_worklist_push(c->wl, child);
    }
}

/* sweep */
static void gc_marksweep_sweep(gc_runtime* rt, GcMarksweepState* state) {
    GcMsMeta** link = &state->objects;
    while (*link != NULL) {
        GcMsMeta* meta = *link;
        if (meta->marked) {
            meta->marked = 0;
            link = &meta->next;
        } else {
            *link = meta->next;
            gc_reclaim_object(rt, ms_obj_of(meta));
            /* 注意：gc_reclaim_object 内部 free 的是 obj 地址，
               但实际 block 起始是 meta 地址 → 见 §2.3 */
        }
    }
}
```

### 2.3 释放的陷阱（极重要）

`gc_reclaim_object(rt, obj)` 内部会 `free(obj)`。但实际 malloc 返回的是 `meta` 地址（block 起始），不是 `obj`。**直接 `free(obj)` 会崩溃**。

两种解法：

**解法 1（推荐）**：算法自己释放，不调用 `gc_reclaim_object` 的 free 部分。
```c
/* 在 sweep 中： */
if (meta 不存活) {
    gc_header* obj = ms_obj_of(meta);
    /* 手动调 finalize（gc_reclaim_object 的前半部分） */
    if (obj->desc && obj->desc->finalize)
        obj->desc->finalize(obj);
    rt->stats.total_allocated -= obj->size;
    free(meta);   /* free block 起始地址，不是 obj */
}
```

**解法 2**：增加 `gc_reclaim_object_block(rt, obj, block_base)` 变体，明确传入 block 起始地址。

> **本文档统一采用解法 1。** 任何用内联头法的算法，sweep/reclaim 时必须 `free(meta)` 而非 `free(obj)`。

### 2.4 rc-cycle 的内联头

/* src/algo/bacon-rajan/gc_alloc.c */

typedef struct GcRcMeta {
    uint32_t          rc;          /* 堆引用计数 */
    uint32_t          gc_ref;      /* trial deletion 工作计数 */
    uint8_t           color;       /* RC_COLOR_* */

    uint8_t           _pad[/* 对齐到 16 的倍数 */];

## 3. 方案 B：转发表法（gen-copy 的 forwarding 用）

**思想**：对象移动时，旧地址→新地址的映射存在一张表里。文档 05 中的 `forward[obj]`、`forwarded(obj)`、`forward_table_insert` 用这里实现。

> **命名映射（05 伪代码 ↔ 本文教学名 ↔ 真实代码名）**：
> 本节用 `gen_forward_*` 作为**教学名**讲解实现。真实代码 `src/algo/gen_copy_ms/young_copy.c` 用的前缀是 `gc_gen_*`。对照表：
>
> | 05 伪代码 | 本节教学名 | 真实代码名（gen_copy_ms） |
> |----------|-----------|------------------------|
> | `forward_table_insert(state, from→to)` | `gen_forward_insert` | （并入 minor 内联逻辑） |
> | `forwarded(obj)` / `forward[obj]` 查找 | `gen_forward_lookup` | `gc_gen_find_old_node` / 内联 |
> | `gc_gen_clear_forwarding(state)` | `gen_forward_clear` | `gc_gen_clear_forwarding` |
> | `gc_gen_add_old_node(rt, state, copy)` | — | `gc_gen_add_old_node`（真实存在） |
>
> 低能力 AI 实现时：**gen-copy-ms 直接照抄真实代码的 `gc_gen_*` 名**；本节的 `gen_forward_*` 是给你理解原理用的，不必照搬到代码里。命名是内部 static 函数，按真实代码一致即可。

### 3.1 链表实现（baseline，对象少时够用）

```c
/* src/algo/gen_copy_ms/young_copy.c */

typedef struct GcGenForward {
    gc_header*           from;
    gc_header*           to;
    struct GcGenForward* next;
} GcGenForward;

/* forwarded(obj) → 查找 */
static gc_header* gen_forward_lookup(GcGenCopyMsState* state, gc_header* obj) {
    for (GcGenForward* it = state->forwarded; it; it = it->next) {
        if (it->from == obj) return it->to;
    }
    return NULL;   /* 未转发 */
}

/* forward_table_insert(state, from → to) */
static int gen_forward_insert(GcGenCopyMsState* state, gc_header* from, gc_header* to) {
    GcGenForward* node = (GcGenForward*)calloc(1, sizeof(GcGenForward));
    if (!node) return 0;
    node->from = from;
    node->to   = to;
    node->next = state->forwarded;
    state->forwarded = node;
    return 1;
}

/* 清空转发表（每轮 GC 后） */
static void gen_forward_clear(GcGenCopyMsState* state) {
    GcGenForward* it = state->forwarded;
    while (it) {
        GcGenForward* next = it->next;
        free(it);
        it = next;
    }
    state->forwarded = NULL;
}
```

### 3.2 转发指针法（高性能，对象被移动时）

**思想**：对象被复制后，把旧对象的**第一个字（desc 字段）**改写为指向新地址 + 一个 tag bit。这样查转发是 O(1)，不需要表。

```c
/* 转发标记：利用指针低位（对象 8 字节对齐 → 低 3 位空闲） */
#define GC_FORWARD_TAG 0x1

/* 把旧对象标记为已转发，并记录新地址 */
static inline void gen_set_forwarded(gc_header* old_obj, gc_header* new_obj) {
    /* 复用 desc 字段存新地址（带 tag） */
    old_obj->desc = (const gc_descriptor*)((uintptr_t)new_obj | GC_FORWARD_TAG);
}

static inline int gen_is_forwarded(gc_header* obj) {
    return ((uintptr_t)obj->desc & GC_FORWARD_TAG) != 0;
}

static inline gc_header* gen_forwarding_address(gc_header* obj) {
    return (gc_header*)((uintptr_t)obj->desc & ~(uintptr_t)GC_FORWARD_TAG);
}
```

> **注意**：转发指针法破坏了 `obj->desc`。只能在"旧对象即将被丢弃"时使用（copying GC 中旧 nursery 整体 reset，所以安全）。**marksweep 不能用这个**（对象原地存活，desc 不能破坏）。

## 4. 方案 C：side-table bitmap 法（marksweep 优化 / regional 用）

**思想**：mark bit 不存对象里，而是存一张独立的位图。对象地址 → 位图索引 通过 `(addr - heap_base) / alignment` 计算。

```c
/* 前提：所有对象在一块连续的 arena 中，按 16 字节对齐 */

/* obj → bit index */
static inline size_t mark_bit_index(gc_arena* arena, gc_header* obj) {
    return ((uintptr_t)obj - (uintptr_t)arena->base) / 16;
}

/* mark */
static inline void mark_set(gc_bitmap* bm, gc_arena* arena, gc_header* obj) {
    gc_bitmap_set(bm, mark_bit_index(arena, obj));
}
static inline int mark_test(gc_bitmap* bm, gc_arena* arena, gc_header* obj) {
    return gc_bitmap_test(bm, mark_bit_index(arena, obj));
}
```

> side-table 法要求对象在连续 arena 中。**只有 arena 分配器（Step E）完成后才能用**。在那之前 marksweep 用方案 A（内联头）。

## 5. 三种方案的选用矩阵

| 算法 | 元数据需求 | 推荐方案 | 文档章节 |
|------|-----------|---------|---------|
| marksweep（baseline） | marked bit | A 内联头 | §2.2 |
| marksweep（优化后） | marked bit | C side-table | §4 |
| gen-copy-ms | forwarding | B 链表 → B 转发指针 | §3 |
| rc-cycle | rc/gc_ref/color | A 内联头 | §2.4 |
| regional | mark/forward per-region | C side-table（per-region bitmap） | §4 |

## 6. 对齐计算（所有方案通用）

```c
/* 向上对齐到 align 的倍数（align 必须是 2 的幂） */
static inline size_t gc_align_up(size_t n, size_t align) {
    return (n + align - 1) & ~(align - 1);
}

/* 内联头大小必须对齐，保证后面 gc_header 对齐到 16 字节 */
#define GC_META_ALIGN 16
/* 实际 meta 结构用 _pad 字段补齐到 GC_META_ALIGN 的倍数 */
```

## 7. Step A 的验收测试

```c
/* tests/test_meta_map_substrate.c */
#include "gc_internal.h"
#include <assert.h>
#include <stdlib.h>

/* 测试内联头互转 */
int main(void) {
    /* 模拟内联头法 */
    size_t meta_size = 16;
    size_t obj_size  = sizeof(gc_header) + 32;
    void* block = calloc(1, meta_size + obj_size);
    assert(block != NULL);

    gc_header* obj = (gc_header*)((uint8_t*)block + meta_size);
    /* obj → meta → obj 往返必须相等 */
    void* meta = (uint8_t*)obj - meta_size;
    gc_header* obj2 = (gc_header*)((uint8_t*)meta + meta_size);
    assert(obj == obj2);

    /* 对齐检查：obj 必须 16 字节对齐 */
    assert(((uintptr_t)obj % 16) == 0 || meta_size % 16 == 0);

    free(block);
    return EXIT_SUCCESS;
}
```

> **重要**：本测试只验证映射算术正确。每个具体算法（marksweep/rc-cycle）的元数据映射在各自的 smoke test 中端到端验证。
