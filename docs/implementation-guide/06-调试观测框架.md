# 06 — 调试观测框架

> GC 算法的正确性极难通过单元测试穷举覆盖。必须有一套**渗透到每个阶段**的运行时调试/观测机制，在开发期间能逐步验证每一步行为。

---

## 1. 设计原则

1. **零成本抽象**：Release 构建中调试代码完全消失（`#ifdef GC_DEBUG` 条件编译）
2. **渗透性**：不是事后追加的 printf，而是框架级基础设施，嵌入每个关键路径
3. **结构化输出**：不是自由格式字符串，而是 `key=value` 格式，可被脚本/工具解析
4. **分级控制**：可按模块、按严重程度过滤，避免信息洪水
5. **断言 + 日志 + 事件流**三层互补

---

## 2. 三层调试机制

### 2.1 Layer 1: 编译期断言（invariant checks）

**已存在**的 `gc_*_check_invariants` 函数族。在 Debug 构建中，应在以下时机自动调用：

```c
// gc_internal.h 中定义：
#ifdef GC_DEBUG
#define GC_ASSERT_INVARIANTS(rt) \
    do { if (!gc_runtime_check_invariants(rt)) { \
        gc_debug_panic(rt, __FILE__, __LINE__, "invariant violation"); \
    } } while(0)
#else
#define GC_ASSERT_INVARIANTS(rt) ((void)0)
#endif
```

**插入点**：
- `gc_alloc_typed` 返回前
- `gc_store_ref` 返回前
- `gc_collect_minor/major/full` 返回前
- 算法内部每个 phase 之间

### 2.2 Layer 2: 结构化事件日志

```c
// gc_internal.h 中定义：

typedef enum {
    GC_EVENT_ALLOC,
    GC_EVENT_STORE_REF,
    GC_EVENT_COLLECT_START,
    GC_EVENT_COLLECT_END,
    GC_EVENT_MARK_OBJECT,
    GC_EVENT_SWEEP_OBJECT,
    GC_EVENT_PROMOTE,
    GC_EVENT_FORWARD,
    GC_EVENT_RECLAIM,
    GC_EVENT_RETAIN,
    GC_EVENT_RELEASE,
    GC_EVENT_PURPLE_PUSH,
    GC_EVENT_TRIAL_DELETE_START,
    GC_EVENT_TRIAL_DELETE_END,
    GC_EVENT_BARRIER_DIRTY,
    GC_EVENT_SAFEPOINT_ENTER,
    GC_EVENT_SAFEPOINT_EXIT,
} gc_event_type;

typedef struct {
    gc_event_type type;
    const char*   file;
    int           line;
    void*         obj;        // 相关对象地址
    void*         obj2;       // 第二相关对象（如 old_value）
    size_t        value;      // 数值参数（如 size、rc 值）
    const char*   detail;     // 可选描述字符串
} gc_event;

// 事件发射宏：
#ifdef GC_DEBUG
#define GC_EMIT_EVENT(rt, ev_type, obj_ptr, val, det) \
    gc_debug_emit_event(rt, ev_type, __FILE__, __LINE__, obj_ptr, NULL, val, det)
#else
#define GC_EMIT_EVENT(rt, ev_type, obj_ptr, val, det) ((void)0)
#endif
```

### 2.3 Layer 3: 运行时 panic 与堆 dump

```c
// 当检测到不可恢复的错误时：
void gc_debug_panic(gc_runtime* rt, const char* file, int line, const char* msg);
// 行为：
//   1. 输出 panic 信息到 stderr
//   2. 如果 GC_DEBUG_DUMP_ON_PANIC：dump 整个堆状态
//   3. abort()

// 堆 dump（调试时手动调用或 panic 时自动触发）：
void gc_debug_dump_heap(gc_runtime* rt, FILE* out);
// 输出：
//   - runtime 配置
//   - heap 统计
//   - 所有对象列表（地址、desc->name、size、flags）
//   - handle 列表
//   - dirty cards
//   - 算法私有状态概要
```

## 3. 实现文件

### 3.1 `src/core/debug.c`（新建）

```c
#include "gc_internal.h"
#include <stdio.h>

#ifdef GC_DEBUG

// 事件级别过滤
static uint32_t gc_debug_event_mask = 0xFFFFFFFF;  // 默认全开

// 输出目标
static FILE* gc_debug_output = NULL;  // NULL = stderr

void gc_debug_init(gc_runtime* rt) {
    (void)rt;
    gc_debug_output = stderr;
    // 可通过环境变量 GC_DEBUG_LOG 重定向到文件
    const char* logfile = getenv("GC_DEBUG_LOG");
    if (logfile) {
        gc_debug_output = fopen(logfile, "w");
        if (!gc_debug_output) gc_debug_output = stderr;
    }
    // 可通过环境变量 GC_DEBUG_MASK 控制事件过滤
    const char* mask_str = getenv("GC_DEBUG_MASK");
    if (mask_str) {
        gc_debug_event_mask = (uint32_t)strtoul(mask_str, NULL, 16);
    }
}

void gc_debug_emit_event(gc_runtime* rt, gc_event_type type,
    const char* file, int line, void* obj, void* obj2, size_t value, const char* detail) {
    if (!(gc_debug_event_mask & (1u << type))) return;
    (void)rt;
    fprintf(gc_debug_output, "[GC] event=%d file=%s:%d obj=%p obj2=%p val=%zu %s\n",
        type, file, line, obj, obj2, value, detail ? detail : "");
}

void gc_debug_panic(gc_runtime* rt, const char* file, int line, const char* msg) {
    fprintf(stderr, "\n!!! GC PANIC at %s:%d: %s\n", file, line, msg);
    gc_debug_dump_heap(rt, stderr);
    abort();
}

void gc_debug_dump_heap(gc_runtime* rt, FILE* out) {
    if (!rt) { fprintf(out, "  rt=NULL\n"); return; }
    fprintf(out, "=== GC HEAP DUMP ===\n");
    fprintf(out, "  algorithm: %s\n", rt->algo ? rt->algo->name : "NULL");
    fprintf(out, "  gc_state: %d\n", GC_ATOMIC_LOAD(&rt->gc_state));
    fprintf(out, "  total_allocated: %zu\n", rt->stats.total_allocated);
    fprintf(out, "  peak_allocated: %zu\n", rt->stats.peak_allocated);
    fprintf(out, "  young: base=%p capacity=%zu used=%zu\n",
        (void*)rt->heap.young_base, rt->heap.young_capacity, rt->heap.young_used);
    fprintf(out, "  old: count=%zu bytes=%zu\n",
        rt->heap.old_object_count, rt->heap.old_object_bytes);
    fprintf(out, "  dirty_old_objects: %zu\n", rt->barriers.dirty_old_objects);
    fprintf(out, "  dirty_cards: %zu\n", rt->barriers.dirty_cards);
    fprintf(out, "  handles:\n");
    for (gc_handle* h = rt->handles; h; h = h->next) {
        fprintf(out, "    handle=%p target=%p flags=%u\n", (void*)h, (void*)h->target, h->flags);
    }
    fprintf(out, "=== END DUMP ===\n");
}

#endif /* GC_DEBUG */
```

## 4. 各模块调试插桩点

### 4.1 runtime.c

```c
gc_runtime* gc_runtime_create(...) {
    ...
    #ifdef GC_DEBUG
    gc_debug_init(rt);
    GC_EMIT_EVENT(rt, GC_EVENT_ALLOC, rt, 0, "runtime_create");
    #endif
    return rt;
}

void gc_store_ref(...) {
    ...
    GC_EMIT_EVENT(rt, GC_EVENT_STORE_REF, owner, (size_t)new_value, NULL);
    GC_ASSERT_INVARIANTS(rt);
}

void gc_collect_full(...) {
    GC_EMIT_EVENT(rt, GC_EVENT_COLLECT_START, NULL, 0, "full");
    ...
    GC_EMIT_EVENT(rt, GC_EVENT_COLLECT_END, NULL, rt->stats.full_collections, "full");
    GC_ASSERT_INVARIANTS(rt);
}
```

### 4.2 marksweep/ms.c

```c
// mark 阶段
static void mark_object(...) {
    GC_EMIT_EVENT(ctx->rt, GC_EVENT_MARK_OBJECT, obj, 0, NULL);
    ...
}

// sweep 阶段
static void sweep(...) {
    ...
    if (!node->marked) {
        GC_EMIT_EVENT(rt, GC_EVENT_SWEEP_OBJECT, node->obj, node->obj->size, NULL);
        gc_reclaim_object(rt, node->obj);
    }
}
```

### 4.3 gen_copy_ms/young_copy.c

```c
static gc_header* promote(...) {
    GC_EMIT_EVENT(ctx->rt, GC_EVENT_PROMOTE, obj, obj->size, NULL);
    ...
    GC_EMIT_EVENT(ctx->rt, GC_EVENT_FORWARD, obj, (size_t)copy, NULL);
}

// dirty card 遍历
static void minor_visit_dirty_card(...) {
    GC_EMIT_EVENT(minor->rt, GC_EVENT_BARRIER_DIRTY, owner, card_index, NULL);
}
```

### 4.4 bacon-rajan/gc_assign.c

```c
// retain
GC_EMIT_EVENT(rt, GC_EVENT_RETAIN, new_value, node->rc + 1, NULL);
// release
GC_EMIT_EVENT(rt, GC_EVENT_RELEASE, old_value, node->rc - 1, NULL);
// push purple
GC_EMIT_EVENT(rt, GC_EVENT_PURPLE_PUSH, old_value, 0, NULL);
```

## 6. 验证调试框架正确性的测试

```c
// tests/test_debug_substrate.c
#ifdef GC_DEBUG

#include "gc_internal.h"
#include <assert.h>

static int event_count = 0;
// 重定向 gc_debug_output 到 /dev/null 或计数器

int main(void) {
    gc_config cfg;
    gc_vm_hooks hooks = {0};
    gc_config_init_default(&cfg);
```

## 7. 使用方法

| 目标 | 命令 |
|------|------|
| 开启调试构建 | `cmake -B build-debug -DGC_DEBUG=ON -DCMAKE_BUILD_TYPE=Debug` |
| 输出到文件 | `GC_DEBUG_LOG=gc.log ./build-debug/tests/test_gen_copy_ms_smoke` |
| 只看 collect 事件 | `GC_DEBUG_MASK=0C ./build-debug/...`（bit 2+3 = COLLECT_START + COLLECT_END） |
| panic 时自动 dump | 默认行为，`gc_debug_panic` 总是调用 `gc_debug_dump_heap` |
| 手动触发 dump | `gc_debug_dump_heap(rt, stdout)` 在任意断点处调用 |
