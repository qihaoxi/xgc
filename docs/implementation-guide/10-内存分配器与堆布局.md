# 10 — 内存分配器与堆布局

> 工业级 GC 不能依赖 malloc/free 作为唯一堆管理。本文件定义 xgc 的多层内存分配架构。

---

## 1. 三层分配架构

```
┌─────────────────────────────────────────────┐
│ Layer 3: GC 分配入口                         │
│   gc_alloc_typed → algo->alloc              │
│   决定对象进哪个 space/generation            │
├─────────────────────────────────────────────┤
│ Layer 2: Space 管理                          │
│   young_space (bump-pointer)                │
│   old_space (size-class freelist)           │
│   large_object_space (直接 mmap/malloc)      │
├─────────────────────────────────────────────┤
│ Layer 1: 页/Arena 分配器                     │
│   gc_arena_alloc / gc_page_alloc            │
│   OS 交互层 (mmap / VirtualAlloc)           │
└─────────────────────────────────────────────┘
```

## 2. Arena 分配器

### 2.1 接口

```c
// src/gc_internal.h
typedef struct gc_arena {
    uint8_t* base;
    size_t   capacity;
    size_t   used;
    size_t   alignment;     // 默认 8
} gc_arena;

void  gc_arena_init(gc_arena* arena, size_t capacity);
void  gc_arena_init_from_buffer(gc_arena* arena, uint8_t* buf, size_t capacity);
void  gc_arena_destroy(gc_arena* arena);
void* gc_arena_alloc(gc_arena* arena, size_t size);
void  gc_arena_reset(gc_arena* arena);  // bump pointer 归零
int   gc_arena_contains(const gc_arena* arena, const void* ptr);
size_t gc_arena_remaining(const gc_arena* arena);
int   gc_arena_check_invariants(const gc_arena* arena);
```

### 2.2 实现（`src/core/arena.c`）

```c
void gc_arena_init(gc_arena* arena, size_t capacity) {
    arena->base = (uint8_t*)malloc(capacity);
    arena->capacity = capacity;
    arena->used = 0;
    arena->alignment = 8;
}

void* gc_arena_alloc(gc_arena* arena, size_t size) {
    size_t aligned = (size + arena->alignment - 1) & ~(arena->alignment - 1);
    if (arena->used + aligned > arena->capacity) return NULL;
    void* ptr = arena->base + arena->used;
    arena->used += aligned;
    return ptr;
}

void gc_arena_reset(gc_arena* arena) {
    arena->used = 0;
    // 注意：不 memset — 对象头由 alloc 时初始化
}
```

## 3. Space 抽象

### 3.1 Young Space（Nursery）

```c
typedef struct gc_young_space {
    gc_arena arena;             // bump-pointer 区域
    size_t   survivor_threshold; // 多大对象不走 nursery
} gc_young_space;

// 分配：gc_arena_alloc（极快）
// 回收：minor GC 后 gc_arena_reset（整体清空）
```

### 3.2 Old Space

```c
typedef struct gc_old_space {
    gc_size_class_allocator sca;   // size-class freelist（小对象）
    gc_header*              list;   // 全对象链表（用于 sweep）
    size_t                  object_count;
    size_t                  total_bytes;
} gc_old_space;

// 分配：size-class freelist（或 fallback malloc）
// 回收：mark-sweep 或 mark-compact
```

### 3.3 Large Object Space (LOS)

```c
typedef struct gc_large_object {
    gc_header*                 obj;
    size_t                     size;
    struct gc_large_object*    next;
} gc_large_object;

typedef struct gc_large_object_space {
    gc_large_object* objects;
    size_t           threshold;     // 超过此大小进入 LOS
    size_t           total_bytes;
} gc_large_object_space;

// 大对象特点：
//   - 永不复制（太贵）
//   - 永不 compact（除非用户显式请求）
//   - mark-sweep 回收
//   - 不进 nursery
```

## 4. 页分配器（OS 交互层）

### 4.1 接口

```c
// src/core/page.c（新建）
void* gc_page_alloc(size_t size);            // mmap 或 malloc
void  gc_page_free(void* ptr, size_t size);  // munmap 或 free
void  gc_page_commit(void* ptr, size_t size);   // 提交物理页

```

### 4.2 跨平台

```c
#ifdef __linux__
    #include <sys/mman.h>
    void* gc_page_alloc(size_t size) {
        return mmap(NULL, size, PROT_READ|PROT_WRITE, MAP_PRIVATE|MAP_ANONYMOUS, -1, 0);
    }
#elif defined(_WIN32)
    void* gc_page_alloc(size_t size) {
        return VirtualAlloc(NULL, size, MEM_RESERVE|MEM_COMMIT, PAGE_READWRITE);
    }
#else
    void* gc_page_alloc(size_t size) {
        return malloc(size);  // fallback
    }
#endif
```

## 5. Size-Class 分配器

### 5.1 数据结构

```c
#define GC_SIZE_CLASS_COUNT 8
static const size_t GC_SIZE_CLASSES[GC_SIZE_CLASS_COUNT] = {
    16, 32, 64, 128, 256, 512, 1024, 2048
};

typedef struct gc_freelist_node {
    struct gc_freelist_node* next;
} gc_freelist_node;

typedef struct gc_size_class {
    gc_freelist_node* freelist;
    size_t            slot_size;
    size_t            allocated_count;
    size_t            free_count;
} gc_size_class;

typedef struct gc_size_class_allocator {
    gc_size_class classes[GC_SIZE_CLASS_COUNT];
} gc_size_class_allocator;
```

### 5.2 分配/释放

```c
void* gc_sca_alloc(gc_size_class_allocator* sca, size_t size) {
    int idx = gc_sca_find_class(size);
    if (idx < 0) return malloc(size);  // 超过最大 class → fallback

    gc_size_class* cls = &sca->classes[idx];
    if (cls->freelist) {
        gc_freelist_node* node = cls->freelist;
        cls->freelist = node->next;
        cls->free_count--;
        cls->allocated_count++;
        return (void*)node;
    }
    // freelist 空 → 从 page allocator 批量补充
    return gc_sca_refill(cls);
}

void gc_sca_free(gc_size_class_allocator* sca, void* ptr, size_t size) {
    int idx = gc_sca_find_class(size);
    gc_size_class* cls = &sca->classes[idx];
    gc_freelist_node* node = (gc_freelist_node*)ptr;
    node->next = cls->freelist;
    cls->freelist = node;
    cls->free_count++;
    cls->allocated_count--;
}
```

**`gc_sca_find_class` / `gc_sca_refill` 实现**（上面用到但未定义，补全）：

```c
/* 找到 >= size 的最小 size class 索引；超过最大 class 返回 -1（调用方 fallback malloc） */
static int gc_sca_find_class(size_t size) {
    for (int i = 0; i < GC_SIZE_CLASS_COUNT; i++) {
        if (size <= GC_SIZE_CLASSES[i]) return i;
    }
    return -1;
}

/* freelist 空时，从 page allocator 申请一页，切成多个 slot 挂到 freelist */
static void* gc_sca_refill(gc_size_class* cls) {
    /* 一次切 N 个 slot（如一页 / slot_size 个） */
    size_t per_page = 64;  /* 简化：固定每批 64 个，或按 PAGE_SIZE/slot_size 算 */
    void* page = gc_page_alloc(per_page * cls->slot_size);
    if (!page) return NULL;
    for (size_t i = 0; i < per_page; i++) {
        gc_freelist_node* node = (gc_freelist_node*)((uint8_t*)page + i * cls->slot_size);
        node->next = cls->freelist;
        cls->freelist = node;
        cls->free_count++;
    }
    /* 返回第一个 slot 给调用方 */
    gc_freelist_node* node = cls->freelist;
    cls->freelist = node->next;
    cls->free_count--;
    cls->allocated_count++;
    return (void*)node;
}
```

> 注意 `gc_sca_refill` 申请的 page 在 free 全部 slot 后是否归还 OS，取决于是否记录 page 来源。简化实现可只挂在 freelist 不归还（碎片代价）；工业级需维护 page 列表，全空时 `gc_page_free`。

## 6. Region-Based 堆（为 Regional Collector 预备）

```c
typedef struct gc_region {
    uint8_t*    base;
    size_t      capacity;        // 固定大小（如 256KB）
    size_t      used;
    uint8_t     generation;      // 0=young, 1=old
    uint8_t     evacuation_candidate; // 是否被选中疏散
    gc_bitmap   mark_bits;       // per-region mark bitmap
    gc_bitmap   pin_bits;        // per-region pin bitmap
} gc_region;

typedef struct gc_region_space {
    gc_region*  regions;
    size_t      region_count;
    size_t      region_capacity;
    size_t      region_size;     // 固定粒度
} gc_region_space;
```

**region 分配 API**（05 §F 的 `copy_to_free_region`、20 §5.2 的 `gc_region_alloc`/`gc_region_undo_alloc` 都指这里，**统一用 `gc_region_*` 名**）：

```c
/* 在 region 内 bump-pointer 分配；越界返回 NULL（不 abort） */
void* gc_region_alloc(gc_region* region, size_t size);
/* 回滚最近一次 alloc（CAS 竞争失败时用，把 used 退回） */
void  gc_region_undo_alloc(gc_region* region, void* ptr, size_t size);
/* 整体重置（疏散完成后回收 region） */
void  gc_region_reset(gc_region* region);
/* 对象地址是否落在该 region */
int   gc_region_contains(const gc_region* region, const void* ptr);
```

> **命名统一**：05 §F 伪代码里的 `copy_to_free_region(live_obj)` 是叙述性写法，实现时就是 `gc_region_alloc(target_region, live_obj->size)` + `memcpy`。20 §5.2 已直接用 `gc_region_alloc`/`gc_region_undo_alloc`。低能力 AI 实现时一律用本节的 `gc_region_*` 名，不要另造 `copy_to_free_region`。

## 7. 实现优先级

| 组件 | 优先级 | 依赖算法 |
|------|--------|---------|
| `gc_arena` | P0 | gen-copy-ms nursery |
| `gc_young_space` | P0 | gen-copy-ms |
| `gc_large_object_space` | P1 | 所有算法（大对象路径） |
| `gc_size_class_allocator` | P2 | marksweep / old gen |
| `gc_page_alloc` (mmap) | P2 | 高性能路径 |
| `gc_region_space` | P3 | regional concurrent |
